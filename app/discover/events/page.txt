/* =========================================================
   HELPERS
========================================================= */

const formatDate = (date) => {
  if (!date || !moment(date).isValid()) return "Date to be announced";
  return moment.utc(date).format("DD MMM YYYY");
};

const getDateParts = (date) => {
  if (!date || !moment(date).isValid()) {
    return { day: "--", month: "TBA" };
  }

  return {
    day: moment.utc(date).format("DD"),
    month: moment.utc(date).format("MMM"),
  };
};

const isFeatured = (event) =>
  event.featured === 1 ||
  event.featured === "1" ||
  event.featured === true;

const getEventDate = (event) => event.event_date || event.date || null;

const getEventDescription = (event) =>
  event.short_description ||
  event.shortDescription ||
  event.description ||
  "Discover our healthcare initiatives and community wellness programs.";

const getEventStatus = (event) => {
  if (event.status) return String(event.status).toLowerCase();

  const date = getEventDate(event);
  if (!date) return "upcoming";

  return moment.utc(date).isBefore(moment.utc(), "day")
    ? "completed"
    : "upcoming";
};

/* =========================================================
   IMAGE WITH FALLBACK
========================================================= */

function EventImage({ event, className = "" }) {
  const [failed, setFailed] = useState(false);

  return (
    <div className={`relative overflow-hidden bg-[#DCEDEF] ${className}`}>
      {event.image && !failed ? (
        <img
          src={event.image}
          alt={event.title || "Hospital event"}
          loading="lazy"
          onError={() => setFailed(true)}
          className="h-full w-full object-cover transition-transform duration-700 ease-out group-hover:scale-[1.06]"
        />
      ) : (
        <div className="flex h-full w-full items-center justify-center bg-gradient-to-br from-[#E4F4F4] to-[#D5E7F0]">
          <HeartPulse size={48} strokeWidth={1.2} className="text-[#087F8C]" />
        </div>
      )}
    </div>
  );
}

                  type="button"
                  onClick={() => setActiveStatus(item.value)}
                  className={`rounded-lg px-3.5 py-2.5 text-[11px] font-bold transition-all duration-200 ${
                    activeStatus === item.value
                      ? "bg-[#073D53] text-white shadow-sm"
                      : "text-slate-500 hover:bg-white hover:text-[#07566B]"
                  }`}
                >
                  {item.label}
                </button>
              ))}
            </div>
          </div>

          {categories.length > 1 && (
            <div className="mt-3 flex flex-wrap items-center gap-2 border-t border-[#EDF1F2] px-1 pt-3">
              <span className="mr-1 text-[10px] font-bold uppercase tracking-wider text-slate-400">
                Categories
              </span>

              {categories.map((category) => (
                <button
                  key={category}
                  type="button"
                  onClick={() => setActiveCategory(category)}
                  className={`rounded-full border px-3 py-1.5 text-[11px] font-semibold transition-all ${
                    activeCategory === category
                      ? "border-[#B8DEDF] bg-[#EAF6F5] text-[#07566B]"
                      : "border-transparent bg-[#F6F8F9] text-slate-500 hover:border-slate-200 hover:bg-white"
                  }`}
                >
                  {category}
                </button>
              ))}
            </div>
          )}
        </div>

        {/* GRID HEADING */}

        <div className="mb-5 flex items-end justify-between gap-3">
          <div>
            <h3 className="text-[20px] font-semibold tracking-[-0.025em] text-[#153847]">
              {activeStatus === "upcoming"
                ? "Upcoming Events"
                : activeStatus === "completed"
                  ? "Past Events"
                  : "All Events"}
            </h3>

            <p className="mt-1 text-[11px] text-slate-500">
              {isLoading
                ? "Finding the latest events..."
                : `${filteredEvents.length} ${
                    filteredEvents.length === 1 ? "event" : "events"
                  } to explore`}
            </p>
          </div>

          {(search ||
            activeCategory !== "All Events" ||
            activeStatus !== "all") && (
            <button
              type="button"
              onClick={clearFilters}
              className="inline-flex items-center gap-1.5 rounded-lg px-3 py-2 text-[11px] font-bold text-[#087F8C] transition hover:bg-[#EAF6F5]"
            >
              <X size={13} />
              Reset
            </button>
          )}
        </div>

        {/* LOADING STATE */}

        {isLoading && (
          <div className="grid grid-cols-1 gap-5 sm:grid-cols-2 lg:grid-cols-3">
            {Array.from({ length: 6 }).map((_, index) => (
              <LoadingCard key={index} />
            ))}
          </div>
        )}

        {/* ERROR STATE */}

        {error && !isLoading && (
          <div className="rounded-[22px] border border-rose-100 bg-white px-6 py-12 text-center shadow-sm">
            <div className="mx-auto flex h-14 w-14 items-center justify-center rounded-2xl bg-rose-50 text-rose-600">
              <RefreshCw size={24} />
            </div>

            <h3 className="mt-4 text-lg font-bold text-[#153847]">
              Events could not be loaded
            </h3>

            <p className="mx-auto mt-2 max-w-lg text-sm leading-6 text-slate-500">
              {error.message || "Please try again in a moment."}
            </p>

            <button
              type="button"
              onClick={() => mutate()}
              className="mt-5 inline-flex items-center gap-2 rounded-xl bg-[#073D53] px-5 py-3 text-xs font-bold text-white transition hover:bg-[#087F8C]"
            >
              <RefreshCw size={15} />
              Try Again
            </button>
          </div>
        )}

        {/* EVENTS GRID */}

        {!isLoading && !error && filteredEvents.length > 0 && (
          <motion.div
            layout
            className="grid grid-cols-1 items-stretch gap-5 sm:grid-cols-2 lg:grid-cols-3"
          >
            <AnimatePresence mode="popLayout">
              {filteredEvents.map((event, index) => (
                <EventCard
                  key={event.id || event.slug || `${event.title}-${index}`}
                  event={event}
                  index={index}
                />
              ))}
            </AnimatePresence>
          </motion.div>
        )}

        {/* EMPTY STATE */}

        {!isLoading && !error && filteredEvents.length === 0 && (
          <div className="rounded-[22px] border border-dashed border-[#CBDADD] bg-white px-5 py-14 text-center">
            <div className="mx-auto flex h-14 w-14 items-center justify-center rounded-2xl bg-[#EAF6F5] text-[#087F8C]">
              <Search size={23} />
            </div>

            <h3 className="mt-4 text-lg font-bold text-[#153847]">
              Nothing here just yet
            </h3>

            <p className="mt-2 text-sm text-slate-500">
              Try another search or clear your filters to see all events.
            </p>

            <button
              type="button"
              onClick={clearFilters}
              className="mt-5 inline-flex items-center gap-2 rounded-xl border border-[#D8E5E7] px-4 py-2.5 text-xs font-bold text-[#07566B] transition hover:bg-[#EDF7F7]"
            >
              Clear filters
              <ArrowRight size={14} />
            </button>
          </div>
        )}
      </section>

      {/* =====================================================
          PREMIUM CTA
      ===================================================== */}

      <section className="px-4 pb-12 sm:px-6 sm:pb-16">
        <div className="relative mx-auto max-w-6xl overflow-hidden rounded-[26px] bg-[#062F46]">
          <div className="pointer-events-none absolute -right-20 -top-36 h-[330px] w-[330px] rounded-full border border-white/[0.08]" />
          <div className="pointer-events-none absolute -right-8 -top-24 h-[240px] w-[240px] rounded-full border border-white/[0.08]" />
          <div className="pointer-events-none absolute -bottom-28 left-[35%] h-56 w-56 rounded-full bg-[#087F8C]/20 blur-3xl" />

          <div className="relative flex flex-col justify-between gap-7 px-6 py-8 sm:px-9 sm:py-10 md:flex-row md:items-center">
            <div className="max-w-[650px]">
              <div className="mb-3 flex items-center gap-2 text-[10px] font-bold uppercase tracking-[0.2em] text-[#9AE2DC]">
                <HeartPulse size={15} />
                Healthcare With Heart
              </div>

              <h2 className="text-[25px] font-semibold leading-tight tracking-[-0.035em] text-white sm:text-[32px]">
                A healthier community starts together.
              </h2>

              <p className="mt-3 max-w-xl text-[13px] leading-6 text-slate-300">
                Connect with our team to learn more about our health
                initiatives, medical programs and upcoming activities.
              </p>
            </div>

            <Link
              href="/contact"
              className="group inline-flex shrink-0 items-center justify-center gap-3 self-start rounded-xl bg-[#C3F0E9] px-5 py-3.5 text-[12px] font-bold text-[#073B4C] transition duration-300 hover:bg-white md:self-center"
            >
              Get In Touch
              <MoveUpRight
                size={16}
                className="transition-transform duration-300 group-hover:translate-x-0.5 group-hover:-translate-y-0.5"
              />
            </Link>
          </div>
        </div>
      </section>
    </main>
  );
}
